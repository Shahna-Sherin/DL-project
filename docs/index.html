import gradio as gr import spaces import torch from transformers import AutoTokenizer, AutoModelForSeq2SeqLM MODEL_NAME = "facebook/nllb-200-distilled-600M" # Load the translation model tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME) model = AutoModelForSeq2SeqLM.from_pretrained(MODEL_NAME) tokenizer.src_lang = "eng_Latn" @spaces.GPU def translate_to_malayalam(text): if not text or not text.strip(): return "" # Run the model on the allocated GPU model.to("cuda") inputs = tokenizer( text.strip(), return_tensors="pt", truncation=True, max_length=256 ).to("cuda") with torch.no_grad(): translated = model.generate( **inputs, forced_bos_token_id=tokenizer.convert_tokens_to_ids("mal_Mlym"), max_new_tokens=128 ) return tokenizer.batch_decode( translated, skip_special_tokens=True )[0] css = """ .gradio-container { background: #f5fbf7 !important; } #main-title { text-align: center; color: #16834a; } #subtitle { text-align: center; color: #52705e; } #translate-panel { border: 1px solid #d7eadc; border-radius: 14px; padding: 18px; background: white; } #translate-btn { background: #16834a; color: white; border-radius: 10px; } """ with gr.Blocks( title="Malayalam AI Translator", css=css ) as demo: gr.Markdown( "# Malayalam AI Translator", elem_id="main-title" ) gr.Markdown( "English to Malayalam · Neural Machine Translation", elem_id="subtitle" ) with gr.Row(): with gr.Column(elem_id="translate-panel"): gr.Markdown("### Translate from · English") english = gr.Textbox( placeholder="Enter English text here...", lines=9, max_lines=15, label="English text" ) counter = gr.Markdown("0 / 2000") clear = gr.Button("Clear text") with gr.Column(elem_id="translate-panel"): gr.Markdown("### Translation · Malayalam") malayalam = gr.Textbox( lines=9, max_lines=15, label="Malayalam translation", interactive=False ) copy = gr.Button("Copy result") translate = gr.Button( "Translate to Malayalam →", elem_id="translate-btn" ) status = gr.Markdown("") english.change( lambda text: f"{len(text or '')} / 2000", inputs=english, outputs=counter ) clear.click( lambda: ("", "", "0 / 2000", ""), outputs=[english, malayalam, counter, status] ) translate.click( fn=translate_to_malayalam, inputs=english, outputs=malayalam, api_name="translate" ) copy.click( fn=None, inputs=malayalam, outputs=None, js="""(text) => { if (text) { navigator.clipboard.writeText(text); } }""" ) demo.launch(show_api=True)
